# install nodejs

sudo curl -fsSL https://deb.nodesource.com/setup_24.x | sudo -E bash -
sudo apt install -y nodejs
sudo npm install pm2 -g

# install Certbot

sudo apt install certbot python3-certbot-apache -y # for apache
sudo apt install certbot python3-certbot-nginx -y # for nginx
sudo certbot --apache -d www.domain.com -d domain.com
sudo certbot --apache -d api.domain.com
sudo certbot --apache -d admin.domain.com

email: admin@gmail.com

# remove certbot
sudo apt remove certbot
